    def get_contact(mobile_no):

        if docs:
            contact = frappe.get_doc('Contact', docs[0].get('name'))
            return contact
        else:
            return None

    def get_customer(contact_name):
        existing_links = frappe.get_all("Dynamic Link", filters={
            'link_doctype': 'Customer',
            'parent': contact_name  # Parent is the Contact document
        }, fields=['link_name'], ignore_permissions=True)

        if not existing_links:
            return None

        return existing_links[0].get('link_name')


    def wa_catch_director(doc, contact, customer):
        """ forward to active ticket, commence whatsapp command action, continue WhatsApp dialogue exchange or prompt user to command """

        def list_customer_issues():
            # Fetch active issues for the customer
            issues = frappe.get_all(
                "Issue",
                filters={"customer": customer, "status": ["!=", "Closed"]},
                fields=["name", "subject", "status"],
                ignore_permissions=True
            )

            if not issues:
                return "No active issues found for the customer."

            # Format the message with each issue in one line
            send_whatsapp_message(contact.mobile_no, "*Active Issues* for Customer: " + customer)
            for issue in issues:
                message = f"*Issue ID: {issue['name']}*\n*Subject:* {issue['subject']}\n*Status:* {issue['status']}"
                send_whatsapp_message(contact.mobile_no, message)


        def send_whatsapp_message(to, message):
            doc = frappe.get_doc({
                'doctype': 'WhatsApp Message',
                'to': to,
                'content_type': 'text',
                'type': 'Outgoing',
                'message': message,
            })
            doc.insert(ignore_permissions=True)

        def extract_command(message):
            # Split the message into words
            words = message.split()

            # Iterate through the words and find the one starting with /issue
            for word in words:
                if word.startswith(('/issue', '/action')):
                    return word

            # If no command is found
            return None

        def create_issue(subject, description, issue_type):
            # frappe.log_error(str(contact), str(contact.name))
            issue = frappe.get_doc({
                'doctype': 'Issue',
                'subject': subject,
                'issue_type': issue_type,
                'description': description,
                'priority': 'High' if issue_type == "Can't Access Server" else 'Low',
                'status': 'Open',
                'customer': customer,
                'custom_contact_chat': contact
            })
            issue.insert(ignore_permissions=True)
            contact.custom_current_support_issue = issue.name
            contact.save(ignore_permissions=True)
            return issue.name

        def forward_to_issue():
            
            def notify_assigned_pupils(issue_doc):
                """Fetches all assigned pupils for a given Issue and sends a system notification."""
                
                issue_name = issue_doc.name
                message = f"WhatsApp from {issue_doc.customer} - {issue_name}"
            
                # Fetch all users assigned to the Issue
                assigned_pupils = frappe.get_all(
                    "ToDo",
                    filters={"reference_type": "Issue", "reference_name": issue_name},
                    fields=["allocated_to"]
                )
            
                if not assigned_pupils:
                    frappe.log_error(f"No pupils assigned to Issue {issue_name}.", "None")
                    return
            
                recipients = [pupil["allocated_to"] for pupil in assigned_pupils]
            
                # Send System Notifications (without imports)
                for recipient in recipients:
                    doc = frappe.new_doc("Notification Log")
                    doc.subject = message
                    doc.content = message
                    doc.for_user = recipient
                    doc.document_type = "Issue"
                    doc.document_name = issue_name
                    doc.insert(ignore_permissions=True)
            
            doc.reference_doctype = 'Issue'
            custom_current_support_issue = contact.get('custom_current_support_issue')
            doc.reference_name = custom_current_support_issue
            doc.save(ignore_permissions=True)
            custom_current_support_issue_doc = frappe.get_doc('Issue', custom_current_support_issue)
            notify_assigned_pupils(custom_current_support_issue_doc)
